     1→// 10X RPC — Rich Presence form (matches Roxy reference exactly)
     2→'use client'
     3→import { useEffect, useState, useRef } from 'react'
     4→import { toast } from 'sonner'
     5→import { api, type RpcConfig } from '@/lib/api-client'
     6→import { PurpleSwitch } from './ui'
     7→import { ACTIVITY_TYPES, PLATFORMS, PLATFORM_GROUPS } from '@/lib/constants'
     8→import { Gamepad2, ChevronDown, Stethoscope, X } from 'lucide-react'
     9→
    10→interface DiagnoseCheck {
    11→  name: string
    12→  ok: boolean
    13→  status: string
    14→  detail?: string
    15→}
    16→interface DiagnoseResult {
    17→  ok: boolean
    18→  user?: string
    19→  checks: DiagnoseCheck[]
    20→  verdict: string
    21→}
    22→
    23→const DEFAULT_CONFIG: RpcConfig = {
    24→  name: '10X RPC',
    25→  type: 'PLAYING',
    26→  platform: 'desktop',
    27→  state: '',
    28→  details: '',
    29→  largeImage: '',
    30→  largeText: '',
    31→  smallImage: '',
    32→  smallText: '',
    33→  button1Label: '',
    34→  button1Url: '',
    35→  button2Label: '',
    36→  button2Url: '',
    37→  partyCurrent: null,
    38→  partyMax: null,
    39→  partyId: '',
    40→  partySecret: '',
    41→  startMinsAgo: 0,
    42→  endTotalMins: null,
    43→  enabled: false,
    44→}
    45→
    46→export function RichPresenceForm({
    47→  initial, onSaved, onToggle, onChange,
    48→}: {
    49→  initial: RpcConfig | null | undefined
    50→  onSaved?: () => void
    51→  onToggle?: (v: boolean) => void
    52→  onChange?: (cfg: RpcConfig) => void
    53→}) {
    54→  const [cfg, setCfg] = useState<RpcConfig>(initial || DEFAULT_CONFIG)
    55→  const [saving, setSaving] = useState(false)
    56→  const [enabled, setEnabled] = useState(initial?.enabled ?? false)
    57→  const [diagnosing, setDiagnosing] = useState(false)
    58→  const [diagnoseResult, setDiagnoseResult] = useState<DiagnoseResult | null>(null)
    59→
    60→  const handleDiagnose = async () => {
    61→    setDiagnosing(true)
    62→    setDiagnoseResult(null)
    63→    try {
    64→      const res = await fetch('/api/rpc/diagnose', { cache: 'no-store' })
    65→      const data: DiagnoseResult = await res.json()
    66→      setDiagnoseResult(data)
    67→      const failing = data.checks.filter(c => !c.ok)
    68→      if (failing.length === 0) {
    69→        toast.success('All checks passed — RPC should be working', { duration: 4000 })
    70→      } else {
    71→        toast.error(`${failing.length} issue(s) found — tap the report to view`, { duration: 6000 })
    72→      }
    73→    } catch (e) {
    74→      toast.error('Diagnose failed: ' + (e instanceof Error ? e.message : 'unknown'))
    75→    } finally {
    76→      setDiagnosing(false)
    77→    }
    78→  }
    79→
    80→  const [platformOpen, setPlatformOpen] = useState(false)
    81→  const [typeOpen, setTypeOpen] = useState(false)
    82→  const platformRef = useRef<HTMLDivElement>(null)
    83→  const typeRef = useRef<HTMLDivElement>(null)
    84→
    85→  useEffect(() => {
    86→    const handleClickOutside = (e: MouseEvent) => {
    87→      const target = e.target as Node
    88→      if (platformRef.current && !platformRef.current.contains(target)) {
    89→        setPlatformOpen(false)
    90→      }
    91→      if (typeRef.current && !typeRef.current.contains(target)) {
    92→        setTypeOpen(false)
    93→      }
    94→    }
    95→    document.addEventListener('mousedown', handleClickOutside)
    96→    return () => document.removeEventListener('mousedown', handleClickOutside)
    97→  }, [])
    98→
    99→  useEffect(() => {
   100→    if (initial) {
   101→      setCfg({
   102→        name: initial.name ?? '10X RPC',
   103→        type: initial.type ?? 'PLAYING',
   104→        platform: initial.platform ?? 'desktop',
   105→        state: initial.state ?? '',
   106→        details: initial.details ?? '',
   107→        largeImage: initial.largeImage ?? '',
   108→        largeText: initial.largeText ?? '',
   109→        smallImage: initial.smallImage ?? '',
   110→        smallText: initial.smallText ?? '',
   111→        button1Label: initial.button1Label ?? '',
   112→        button1Url: initial.button1Url ?? '',
   113→        button2Label: initial.button2Label ?? '',
   114→        button2Url: initial.button2Url ?? '',
   115→        partyCurrent: initial.partyCurrent ?? null,
   116→        partyMax: initial.partyMax ?? null,
   117→        partyId: initial.partyId ?? '',
   118→        partySecret: initial.partySecret ?? '',
   119→        startMinsAgo: initial.startMinsAgo ?? 0,
   120→        endTotalMins: initial.endTotalMins ?? null,
   121→        enabled: initial.enabled ?? false,
   122→      })
   123→      setEnabled(initial.enabled ?? false)
   124→    }
   125→  }, [initial])
   126→

   128→    setCfg(prev => {
   129→      const next = { ...prev, [key]: value }
   130→      onChange?.(next)
   131→      return next
   132→    })
   133→  }
   134→
   135→  const handleSave = async () => {
   136→    const errors: string[] = []
   137→    if ((cfg.name || '').length > 128) errors.push('Name too long (max 128)')
   138→    if ((cfg.state || '').length > 128) errors.push('State too long (max 128)')
   139→    if ((cfg.details || '').length > 128) errors.push('Details too long (max 128)')
   140→    if ((cfg.button1Label || '').length > 32) errors.push('Button 1 label too long (max 32)')
   141→    if ((cfg.button2Label || '').length > 32) errors.push('Button 2 label too long (max 32)')
   142→    if (cfg.button1Url && !/^https?:\/\//i.test(cfg.button1Url)) errors.push('Button 1 URL must start with http:// or https://')
   143→    if (cfg.button2Url && !/^https?:\/\//i.test(cfg.button2Url)) errors.push('Button 2 URL must start with http:// or https://')
   144→    if (cfg.button1Label && !cfg.button1Url) errors.push('Button 1 URL required when label is set')
   145→    if (cfg.button2Label && !cfg.button2Url) errors.push('Button 2 URL required when label is set')
   146→    if (cfg.partyCurrent != null && cfg.partyMax != null && cfg.partyCurrent > cfg.partyMax) {
   147→      errors.push('Party size cannot exceed party max')
   148→    }
   149→    if (cfg.startMinsAgo != null && cfg.startMinsAgo < 0) errors.push('Start time cannot be negative')
   150→    if (cfg.endTotalMins != null && cfg.endTotalMins < 1) errors.push('End time must be at least 1 minute')

   152→      errors.forEach(e => toast.error(e))
   153→      return

   158→      const res = await api.rpcSave({ ...cfg, enabled })

   160→        toast.success(enabled ? 'Rich Presence updated & live on Discord' : 'Configuration saved (RPC is OFF)', { duration: 2500 })
   161→        onSaved?.()
   162→      } else {
   163→        toast.error('Failed to save configuration')
   164→      }
   165→    } catch (e) {
   166→      console.error(e)
   167→      toast.error('Failed to save Rich Presence')
   168→    } finally {
   169→      setSaving(false)
   170→    }
   171→  }
   172→
   173→  const handleToggle = async (v: boolean) => {
   174→    setEnabled(v)
   175→    try {
   176→      const res = await api.rpcToggle(v)
   177→      if (res.ok) {
   178→        toast.success(v ? 'RPC enabled & live on Discord' : 'RPC stopped & cleared from Discord', { duration: 2500 })
   179→        onToggle?.(v)
   180→        onSaved?.()
   181→      } else {
   182→        setEnabled(!v)
   183→        toast.error('Failed to toggle RPC')
   184→      }
   185→    } catch (e) {
   186→      console.error(e)
   187→      setEnabled(!v)
   188→      toast.error('Failed to toggle RPC')
   189→    }
   190→  }
   191→
   192→  return (
   193→    <div className="space-y-8">
   194→      {/* === RICH PRESENCE CARD (Matches roxydev.xyz reference) === */}
   195→      <div className="relative overflow-hidden bg-gradient-to-b from-[#13111d]/95 via-[#0e0d14]/95 to-[#0a0a0f] border border-white/10 rounded-[28px] p-6 sm:p-7 shadow-2xl backdrop-blur-xl">
   196→        {/* Ambient violet glow at top */}
   197→        <div className="absolute -top-16 left-1/4 -translate-x-1/2 w-64 h-48 bg-purple-600/15 rounded-full blur-3xl pointer-events-none" />
   198→        <div className="absolute -top-16 right-0 w-48 h-48 bg-purple-900/10 rounded-full blur-2xl pointer-events-none" />
   199→
   200→        {/* Centered Card Title: 🎮 Rich Presence */}
   201→        <div className="relative z-10 flex items-center justify-center gap-2.5 mb-7 pt-1">
   202→          <Gamepad2 className="w-6 h-6 text-purple-400 stroke-[2.2]" />

   204→        </div>
   205→
   206→        {/* Form Fields */}
   207→        <div className="relative z-10 space-y-4">
   208→          {/* 1. ACTIVITY TYPE */}
   209→          <FormField label="ACTIVITY TYPE">
   210→            <div className="relative" ref={typeRef}>
   211→              <button
   212→                type="button"
   213→                onClick={() => { setTypeOpen(v => !v); setPlatformOpen(false) }}
   214→                className="w-full h-12 bg-[#12131a] border border-white/10 hover:border-white/20 focus:border-purple-500/50 rounded-xl px-4 text-sm text-white flex items-center justify-between cursor-pointer uppercase font-medium transition-colors"
   215→              >
   216→                <span>{cfg.type || 'PLAYING'}</span>
   217→                <ChevronDown className={`w-4 h-4 text-white/50 transition-transform ${typeOpen ? 'rotate-180' : ''}`} />
   218→              </button>
   219→
   220→              {typeOpen && (
   221→                <div className="absolute left-0 top-full mt-2 w-full bg-[#161720]/98 backdrop-blur-xl border border-white/10 rounded-2xl p-1.5 shadow-2xl z-50 space-y-1">
   222→                  {ACTIVITY_TYPES.map(t => {
   223→                    const isSelected = (cfg.type || 'PLAYING') === t.label
   224→                    return (
   225→                      <button
   226→                        key={t.label}
   227→                        type="button"
   228→                        onClick={() => { set('type', t.label); setTypeOpen(false) }}
   229→                        className={`w-full flex items-center gap-2 px-3 py-2.5 rounded-xl text-xs font-medium uppercase transition-colors text-left ${
   230→                          isSelected
   231→                            ? 'bg-[#6b21a8] text-white font-semibold'
   232→                            : 'text-white/80 hover:text-white hover:bg-white/5'
   233→                        }`}
   234→                      >
   235→                        <span>{t.label}</span>
   236→                      </button>
   237→                    )
   238→                  })}
   239→                </div>
   240→              )}
   241→            </div>
   242→          </FormField>
   243→
   244→          {/* 2. RPC DEVICE / PLATFORM */}
   245→          <FormField label="RPC DEVICE / PLATFORM">
   246→            <div className="relative" ref={platformRef}>
   247→              <button
   248→                type="button"
   249→                onClick={() => { setPlatformOpen(v => !v); setTypeOpen(false) }}
   250→                className="w-full h-12 bg-[#12131a] border border-white/10 hover:border-white/20 focus:border-purple-500/50 rounded-xl px-4 text-sm text-white flex items-center justify-between cursor-pointer transition-colors"
   251→              >
   252→                {(() => {
   253→                  const currentPlatform = PLATFORMS.find(p => p.value === cfg.platform)
   254→                  if (!cfg.platform || cfg.platform === 'desktop') {
   255→                    return <span className="text-white/90 font-medium">None</span>
   256→                  }
   257→                  return (
   258→                    <span className="flex items-center gap-2 text-white/90 font-medium">
   259→                      <span>{currentPlatform?.emoji || '🖥️'}</span>
   260→                      <span>{currentPlatform?.label || cfg.platform}</span>
   261→                    </span>
   262→                  )
   263→                })()}
   264→                <ChevronDown className={`w-4 h-4 text-white/50 transition-transform ${platformOpen ? 'rotate-180' : ''}`} />
   265→              </button>
   266→
   267→              {platformOpen && (
   268→                <div className="absolute left-0 top-full mt-2 w-full max-h-80 overflow-y-auto bg-[#161720]/98 backdrop-blur-xl border border-white/10 rounded-2xl p-2 shadow-2xl z-50 space-y-2">
   269→                  {/* None option */}
   270→                  <button
   271→                    type="button"
   272→                    onClick={() => { set('platform', 'desktop'); setPlatformOpen(false) }}
   273→                    className={`w-full flex items-center gap-2 px-3 py-2 rounded-xl text-xs transition-colors text-left ${
   274→                      (!cfg.platform || cfg.platform === 'desktop')
   275→                        ? 'bg-[#6b21a8] text-white font-medium'
   276→                        : 'text-white/80 hover:text-white hover:bg-white/5'
   277→                    }`}
   278→                  >
   279→                    <span>None</span>
   280→                  </button>
   281→
   282→                  {/* Grouped Platform Categories */}
   283→                  {PLATFORM_GROUPS.map(group => (
   284→                    <div key={group} className="mb-1.5">
   285→                      <p className="text-[10px] uppercase tracking-wider text-white/40 px-2 py-1 font-semibold">{group}</p>
   286→                      <div className="space-y-0.5">
   287→                        {PLATFORMS.filter(p => p.group === group).map(p => {
   288→                          const isSelected = cfg.platform === p.value
   289→                          return (
   290→                            <button
   291→                              key={p.value}
   292→                              type="button"
   293→                              onClick={() => { set('platform', p.value); setPlatformOpen(false) }}
   294→                              className={`w-full flex items-center gap-2 px-2.5 py-1.5 rounded-lg text-xs transition-colors text-left ${
   295→                                isSelected
   296→                                  ? 'bg-[#6b21a8] text-white font-medium shadow-sm'
   297→                                  : 'text-white/80 hover:text-white hover:bg-white/5'
   298→                              }`}
   299→                            >
   300→                              <span>{p.emoji}</span>
   301→                              <span>{p.label}</span>
   302→                            </button>
   303→                          )
   304→                        })}
   305→                      </div>
   306→                    </div>
   307→                  ))}
   308→                </div>
   309→              )}
   310→            </div>
   311→          </FormField>
   312→
   313→          {/* 3. NAME */}
   314→          <FormField label="NAME">
   315→            <input
   316→              type="text"
   317→              value={cfg.name || ''}
   318→              onChange={e => set('name', e.target.value)}
   319→              placeholder="e.g. Visual Studio Code"
   320→              className="w-full h-12 bg-[#12131a] border border-white/10 hover:border-white/20 focus:border-purple-500/50 rounded-xl px-4 text-sm text-white placeholder:text-white/35 outline-none transition-colors"
   321→              maxLength={128}
   322→            />
   323→          </FormField>
   324→
   325→          {/* 4. STATE */}
   326→          <FormField label="STATE">
   327→            <input
   328→              type="text"
   329→              value={cfg.state || ''}
   330→              onChange={e => set('state', e.target.value)}
   331→              placeholder="e.g. Editing page.tsx"
   332→              className="w-full h-12 bg-[#12131a] border border-white/10 hover:border-white/20 focus:border-purple-500/50 rounded-xl px-4 text-sm text-white placeholder:text-white/35 outline-none transition-colors"
   333→              maxLength={128}
   334→            />
   335→          </FormField>
   336→
   337→          {/* 5. DETAILS */}
   338→          <FormField label="DETAILS">
   339→            <input

   342→              onChange={e => set('details', e.target.value)}
   343→              placeholder="e.g. Workspace: Project"
   344→              className="w-full h-12 bg-[#12131a] border border-white/10 hover:border-white/20 focus:border-purple-500/50 rounded-xl px-4 text-sm text-white placeholder:text-white/35 outline-none transition-colors"
   345→              maxLength={128}

   391→              maxLength={128}

   484→            <input

   486→              value={cfg.partySecret || ''}
   487→              onChange={e => set('partySecret', e.target.value)}
   488→              placeholder="e.g. secret-456"
   489→              className="w-full h-12 bg-[#12131a] border border-white/10 hover:border-white/20 focus:border-purple-500/50 rounded-xl px-4 text-sm text-white placeholder:text-white/35 outline-none transition-colors"
   490→            />
   491→          </FormField>
   492→
   493→          {/* 18. START TIME (MINS AGO) */}
   494→          <FormField label="START TIME (MINS AGO)">
   495→            <input
   496→              type="number"
   497→              min="0"
   498→              value={cfg.startMinsAgo ?? ''}
   499→              onChange={e => set('startMinsAgo', e.target.value ? Number(e.target.value) : 0)}
   500→              placeholder="1"
   501→              className="w-full h-12 bg-[#12131a] border border-white/10 hover:border-white/20 focus:border-purple-500/50 rounded-xl px-4 text-sm text-white placeholder:text-white/35 outline-none transition-colors"
   502→            />
   503→          </FormField>
   504→
   505→          {/* 19. END TIME (TOTAL MINS FROM START) */}
   506→          <FormField label="END TIME (TOTAL MINS FROM START)">
   507→            <input
   508→              type="number"
   509→              min="1"
   510→              value={cfg.endTotalMins ?? ''}
   511→              onChange={e => set('endTotalMins', e.target.value ? Number(e.target.value) : null)}
   512→              placeholder="e.g. 34"
   513→              className="w-full h-12 bg-[#12131a] border border-white/10 hover:border-white/20 focus:border-purple-500/50 rounded-xl px-4 text-sm text-white placeholder:text-white/35 outline-none transition-colors"
   514→            />
   515→          </FormField>
   516→
   517→          {/* Divider line before ENABLE RPC */}
   518→          <div className="border-t border-white/10 pt-6 mt-6">
   519→            {/* ENABLE RPC Toggle Row */}
   520→            <div className="flex items-center justify-between px-1 mb-5">
   521→              <span className="text-sm sm:text-base font-extrabold tracking-widest text-[#a855f7] uppercase">
   522→                ENABLE RPC
   523→              </span>
   524→              <PurpleSwitch checked={enabled} onCheckedChange={handleToggle} />
   525→            </div>
   526→
   527→            {/* Centered UPDATE + Diagnose Buttons */}
   528→            <div className="flex justify-center gap-2 flex-wrap">
   529→              <button
   530→                type="button"
   531→                onClick={handleSave}
   532→                disabled={saving}
   533→                className="bg-[#1e1f26] hover:bg-[#282933] border border-white/10 text-white font-medium text-xs tracking-widest uppercase px-8 py-3 rounded-xl transition-all active:scale-[0.98] disabled:opacity-50"
   534→              >
   535→                {saving ? 'SAVING...' : 'UPDATE'}
   536→              </button>
   537→              <button
   538→                type="button"
   539→                onClick={handleDiagnose}
   540→                disabled={diagnosing}
   541→                className="bg-[#1e1f26] hover:bg-[#282933] border border-purple-500/30 text-purple-200 font-medium text-xs tracking-widest uppercase px-4 py-3 rounded-xl transition-all active:scale-[0.98] disabled:opacity-50 inline-flex items-center gap-1.5"
   542→                title="Run a full diagnostic on why RPC may not be showing on Discord"
   543→              >
   544→                <Stethoscope className="w-3.5 h-3.5" />
   545→                {diagnosing ? 'CHECKING...' : 'DIAGNOSE'}
   546→              </button>
   547→            </div>
   548→
   549→            {/* Diagnose Results */}
   550→            {diagnoseResult && (
   551→              <div className="mt-4 p-4 rounded-xl bg-[#0f0f14] border border-white/10 max-h-80 overflow-y-auto styled-scroll">
   552→                <div className="flex items-center justify-between mb-3">
   553→                  <span className={`text-xs font-bold uppercase tracking-wider ${diagnoseResult.ok ? 'text-emerald-400' : 'text-red-400'}`}>
   554→                    {diagnoseResult.ok ? '✓ ALL CHECKS PASSED' : `✗ ${diagnoseResult.checks.filter(c => !c.ok).length} ISSUE(S) FOUND`}
   555→                  </span>
   556→                  <button onClick={() => setDiagnoseResult(null)} className="text-white/40 hover:text-white">
   557→                    <X className="w-4 h-4" />
   558→                  </button>
   559→                </div>
   560→                <div className="space-y-2">
   561→                  {diagnoseResult.checks.map((c, i) => (
   562→                    <div key={i} className={`p-2.5 rounded-lg border ${c.ok ? 'border-emerald-500/20 bg-emerald-500/5' : 'border-red-500/30 bg-red-500/8'}`}>
   563→                      <div className="flex items-start gap-2">
   564→                        <span className={`text-sm font-bold shrink-0 ${c.ok ? 'text-emerald-400' : 'text-red-400'}`}>{c.ok ? '✓' : '✗'}</span>
   565→                        <div className="min-w-0 flex-1">
   566→                          <div className="text-xs font-semibold text-white">{c.name}</div>
   567→                          <div className="text-xs text-white/70 mt-0.5 break-words">{c.status}</div>
   568→                          {c.detail && <div className="text-[11px] text-white/40 mt-1 break-words">{c.detail}</div>}
   569→                        </div>
   570→                      </div>
   571→                    </div>
   572→                  ))}
   573→                </div>
   574→                <div className="mt-3 p-2.5 rounded-lg bg-white/5 text-xs text-white/60 leading-relaxed">
   575→                  <span className="font-semibold text-white/80">Verdict:</span> {diagnoseResult.verdict}
   576→                </div>
   577→              </div>
   578→            )}
   579→          </div>
   580→        </div>
   581→      </div>
   582→    </div>
   583→  )
   584→}
   585→

   590→        {label}
   591→      </label>
   592→      {children}
   593→    </div>
   594→  )
   595→}
