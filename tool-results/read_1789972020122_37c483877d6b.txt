    84→    84→  /**
    85→    85→   * Background tick loop:
    86→    86→   * - Rotates custom status if rotator is enabled
    87→    87→   * - Checks and expires sleep timers
    88→    88→   * - Re-resolves dynamic placeholders ({time}, {uptime}, etc.)
    89→    89→   * - Refreshes near-expiry Discord tokens
    90→    90→   * - Syncs any DB state changes
    91→    91→   */
    92→    92→  public async tick(): Promise<void> {
    93→    93→    this.lastTickAt = new Date()
    94→    94→    const now = new Date()
    95→    95→
    96→    96→    // 1. Sync users from DB
    97→    97→    await this.syncAllUsers()
    98→    98→
    99→    99→    // 2. Iterate connected users
   100→   100→    for (const [userId, userSock] of this.sockets.entries()) {
   101→   101→      try {
   102→   102→        const session = await db.session.findFirst({
   103→   103→          where: { userId, expiresAt: { gt: now } },
   104→   104→          include: {
   105→   105→            user: {
   106→   106→              include: {
   107→   107→                globalConfig: true,
   108→   108→                rotatorPresets: { orderBy: { order: 'asc' } },
   109→   109→                trial: true,
   110→   110→              },
   111→   111→            },
   112→   112→          },
   113→   113→        })
   114→   114→
   115→   115→        if (!session) {
   116→   116→          this.disconnectUser(userId)
   117→   117→          continue
   118→   118→        }
   119→   119→
   120→   120→        // Check trial
   121→   121→        if (!session.user.trial || !session.user.trial.active || session.user.trial.endsAt < now) {
   122→   122→          console.log(`[10X RPC Daemon] Trial expired for user ${userId}, disabling presence`)
   123→   123→          await db.session.update({
   124→   124→            where: { id: session.id },
   125→   125→            data: { rpcEnabled: false, statusEnabled: false, gatewayReady: false },
   126→   126→          })
   127→   127→          this.disconnectUser(userId)
   128→   128→          continue
   129→   129→        }
   130→   130→
   131→   131→        // Check sleep timer
   132→   132→        if (session.sleepTimerActive && session.sleepTimerEndsAt && session.sleepTimerEndsAt <= now) {
   133→   133→          console.log(`[10X RPC Daemon] Sleep timer elapsed for user ${userId}`)
   134→   134→          await db.session.update({
   135→   135→            where: { id: session.id },
   136→   136→            data: {
   137→   137→              sleepTimerActive: false,
   138→   138→              sleepTimerEndsAt: null,
   139→   139→              rpcEnabled: false,
   140→   140→              statusEnabled: false,
   141→   141→              gatewayReady: false,
   142→   142→            },
   143→   143→          })
   144→   144→          await db.rpcConfig.updateMany({
   145→   145→            where: { userId, enabled: true },
   146→   146→            data: { enabled: false },
   147→   147→          })
   148→   148→          this.disconnectUser(userId)
   149→   149→          continue
   150→   150→        }
   151→   151→
   152→   152→        // Check token expiration (refresh if < 30 mins left)
   153→   153→        if (session.discordTokenExpiresAt && session.discordTokenExpiresAt.getTime() - now.getTime() < 30 * 60 * 1000) {
   154→   154→          if (session.discordRefreshToken) {
   155→   155→            console.log(`[10X RPC Daemon] Refreshing Discord token for user ${userId}...`)
   156→   156→            const refreshed = await refreshDiscordToken(session.discordRefreshToken)
   157→   157→            if (refreshed) {
   158→   158→              await db.session.update({
   159→   159→                where: { id: session.id },
   160→   160→                data: {
   161→   161→                  discordAccessToken: refreshed.access_token,
   162→   162→                  discordRefreshToken: refreshed.refresh_token,
   163→   163→                  discordTokenExpiresAt: new Date(Date.now() + (refreshed.expires_in || 604800) * 1000),
   164→   164→                },
   165→   165→              })
   166→   166→              session.discordAccessToken = refreshed.access_token
   167→   167→            }
   168→   168→          }
   169→   169→        }
   170→   170→
   171→   171→        // Check status rotator
   172→   172→        if (session.user.globalConfig?.rotatorEnabled) {
   173→   173→          const presets = session.user.rotatorPresets.filter(p => p.enabled)
   174→   174→          if (presets.length > 0) {
   175→   175→            const totalDurationSecs = presets.reduce((sum, p) => sum + Math.max(1, p.durationMins * 60), 0)
   176→   176→            const referenceTime = presets[0].createdAt.getTime()
   177→   177→            const elapsedSecs = Math.floor((now.getTime() - referenceTime) / 1000)
   178→   178→            const positionInCycle = ((elapsedSecs % totalDurationSecs) + totalDurationSecs) % totalDurationSecs
   179→   179→
   180→   180→            let accumulated = 0
   181→   181→            let activeIndex = 0
   182→   182→            for (let i = 0; i < presets.length; i++) {
   183→   183→              const dur = Math.max(1, presets[i].durationMins * 60)
   184→   184→              if (accumulated + dur > positionInCycle) {
   185→   185→                activeIndex = i
   186→   186→                break
   187→   187→              }
   188→   188→              accumulated += dur
   189→   189→            }
   190→   190→
   191→   191→            const activePreset = presets[activeIndex]
   192→   192→            if (session.customStatus !== activePreset.text || session.customStatusEmoji !== activePreset.emoji) {
   193→   193→              await db.session.update({
   194→   194→                where: { id: session.id },
   195→   195→                data: {
   196→   196→                  customStatus: activePreset.text,
   197→   197→                  customStatusEmoji: activePreset.emoji,
   198→   198→                },
   199→   199→              })
   200→   200→              session.customStatus = activePreset.text
   201→   201→              session.customStatusEmoji = activePreset.emoji
   202→   202→            }
   203→   203→          }
   204→   204→        }
   205→   205→
   206→   206→        // Re-evaluate activities and push OP 3 if needed
   207→   207→        await this.pushPresenceForUser(userId, session)
   208→   208→      } catch (userTickErr) {
   209→   209→        console.error(`[10X RPC Daemon] Error in tick for user ${userId}:`, userTickErr)
   210→   210→      }
   211→   211→    }
   212→   212→  }

   214→   214→  /**
   215→   215→   * Synchronize all active users from Postgres DB.
   216→   216→   * Only keeps connections alive for users with active RPC, custom status, or status rotator.
   217→   217→   * Cleans up and disconnects any user whose RPC and status are disabled.
   218→   218→   */
   219→   219→  public async syncAllUsers(): Promise<void> {
   220→   220→    const now = new Date()
   221→   221→    const activeSessions = await db.session.findMany({
   222→   222→      where: {
   223→   223→        expiresAt: { gt: now },
   224→   224→        discordAccessToken: { not: null },
   225→   225→      },
   226→   226→      include: {
   227→   227→        user: {
   228→   228→          include: {
   229→   229→            trial: true,
   230→   230→            globalConfig: true,
   231→   231→            rpcConfigs: true,
   232→   232→            rotatorPresets: true,
   233→   233→          },
   234→   234→        },
   235→   235→      },
   236→   236→    })
   237→   237→
   238→   238→    const activeUserIds = new Set<string>()
   239→   239→
   240→   240→    for (const session of activeSessions) {
   241→   241→      // Check trial
   242→   242→      if (!session.user.trial || !session.user.trial.active || session.user.trial.endsAt < now) {
   243→   243→        continue
   244→   244→      }
   245→   245→      // Check sleep timer
   246→   246→      if (session.sleepTimerActive && session.sleepTimerEndsAt && session.sleepTimerEndsAt <= now) {
   247→   247→        continue
   248→   248→      }
   249→   249→
   250→   250→      const rpcConfig = session.user.rpcConfigs?.[0]
   251→   251→      const hasRpc = !!(session.rpcEnabled && rpcConfig?.enabled)
   252→   252→      const hasStatus = !!session.statusEnabled
   253→   253→      const hasRotator = !!(session.statusEnabled && session.user.globalConfig?.rotatorEnabled && session.user.rotatorPresets?.some(p => p.enabled))
   254→   254→
   255→   255→      // Only track if user actually has active RPC, active status, or active rotator
   256→   256→      if (!hasRpc && !hasStatus && !hasRotator) {
   257→   257→        continue
   258→   258→      }
   259→   259→
   260→   260→      activeUserIds.add(session.userId)
   261→   261→      let userSock = this.sockets.get(session.userId)
   262→   262→
   263→   263→      if (!userSock) {
   264→   264→        userSock = {
   265→   265→          userId: session.userId,
   266→   266→          sessionId: session.id,
   267→   267→          ws: null,
   268→   268→          heartbeatAck: true,
   269→   269→          retryCount: 0,
   270→   270→          platform: rpcConfig?.platform || 'desktop',
   271→   271→          lastStatus: session.userStatus || 'online',
   272→   272→          lastActivitiesHash: '',
   273→   273→          connected: false,
   274→   274→          isConnecting: false,
   275→   275→        }
   276→   276→        this.sockets.set(session.userId, userSock)
   277→   277→      }
   278→   278→
   279→   279→      // If socket is disconnected, connect it
   280→   280→      if (!userSock.connected && !userSock.isConnecting) {
   281→   281→        this.connectUserSocket(session.userId)
   282→   282→      }
   283→   283→    }
   284→   284→
   285→   285→    // Cleanup and disconnect users that are no longer active (prevents background restarts)
   286→   286→    for (const [userId, userSock] of this.sockets.entries()) {
   287→   287→      if (!activeUserIds.has(userId)) {
   288→   288→        this.disconnectUser(userId)
   289→   289→      }
   290→   290→    }
   291→   291→  }

   293→   293→  /**
   294→   294→   * Stop RPC completely for a user:
   295→   295→   * - Sends OP 3 with empty activities (or custom status only) to clear Discord Rich Presence
   296→   296→   * - Stops all timers and intervals
   297→   297→   * - Disconnects socket if no other presence (e.g. custom status) is needed
   298→   298→   * - Prevents background restarts
   299→   299→   */
   300→   300→  public async stopUserRpc(userId: string): Promise<void> {
   301→   301→    const session = await db.session.findFirst({
   302→   302→      where: { userId },
   303→   303→    })
   304→   304→    if (!session || !session.discordAccessToken) return
   305→   305→
   306→   306→    const userSock = this.sockets.get(userId)
   307→   307→    const hasStatus = !!session.statusEnabled
   308→   308→    const hasCustomStatus = !!(session.customStatus || session.customStatusEmoji)
   309→   309→
   310→   310→    if (userSock && userSock.ws && userSock.ws.readyState === WebSocket.OPEN) {
   311→   311→      const activities: Array<Record<string, unknown>> = []
   312→   312→      if (hasStatus && hasCustomStatus) {
   313→   313→        const customActivity = buildCustomStatusActivity(
   314→   314→          session.customStatus,
   315→   315→          session.customStatusEmoji
   316→   316→        )
   317→   317→        if (customActivity) activities.push(customActivity)
   318→   318→      }
   319→   319→
   320→   320→      const status = hasStatus ? (session.userStatus || 'online') : 'invisible'
   321→   321→
   322→   322→      try {
   323→   323→        userSock.ws.send(JSON.stringify({
   324→   324→          op: 3,
   325→   325→          d: {
   326→   326→            status,
   327→   327→            activities,
   328→   328→            afk: false,
   329→   329→            since: null,
   330→   330→          },
   331→   331→        }))
   332→   332→        userSock.lastActivitiesHash = JSON.stringify({ status, activities })
   333→   333→        userSock.lastStatus = status
   334→   334→      } catch (err) {
   335→   335→        console.error(`[10X RPC Daemon] Error sending clear OP 3 for user ${userId}:`, err)
   336→   336→      }
   337→   337→
   338→   338→      // If status is not active, close socket cleanly and stop all timers
   339→   339→      if (!hasStatus) {
   340→   340→        this.cleanupSocket(userSock)
   341→   341→        this.sockets.delete(userId)
   342→   342→      }
   343→   343→    } else if (hasStatus) {
   344→   344→      // Connect to Discord Gateway to maintain user status
   345→   345→      await this.connectUserSocket(userId)
   346→   346→    }
   347→   347→  }

   431→   431→  /**
   432→   432→   * Connect or reconnect a user's WebSocket to Discord Gaming SDK Gateway.
   433→   433→   */
   434→   434→  private async connectUserSocket(userId: string): Promise<void> {
   435→   435→    const userSock = this.sockets.get(userId)
   436→   436→    if (!userSock || userSock.isConnecting) return
   437→   437→
   438→   438→    // Clean up previous socket cleanly before starting new connection
   439→   439→    this.cleanupSocket(userSock)
   440→   440→    userSock.isConnecting = true
   441→   441→
   442→   442→    try {
   443→   443→      const session = await db.session.findFirst({
   444→   444→        where: { userId },
   445→   445→      })
   446→   446→      if (!session || !session.discordAccessToken) {
   447→   447→        userSock.isConnecting = false
   448→   448→        this.disconnectUser(userId)
   449→   449→        return
   450→   450→      }
   451→   451→
   452→   452→      // Check if token is expired and refresh before connecting
   453→   453→      let accessToken = session.discordAccessToken
   454→   454→      const now = new Date()
   455→   455→      if (session.discordTokenExpiresAt && session.discordTokenExpiresAt < now) {
   456→   456→        if (session.discordRefreshToken) {
   457→   457→          const refreshed = await refreshDiscordToken(session.discordRefreshToken)
   458→   458→          if (refreshed) {
   459→   459→            accessToken = refreshed.access_token
   460→   460→            await db.session.update({
   461→   461→              where: { id: session.id },
   462→   462→              data: {
   463→   463→                discordAccessToken: refreshed.access_token,
   464→   464→                discordRefreshToken: refreshed.refresh_token,
   465→   465→                discordTokenExpiresAt: new Date(Date.now() + (refreshed.expires_in || 604800) * 1000),
   466→   466→              },
   467→   467→            })
   468→   468→          }
   469→   469→        }
   470→   470→      }
   471→   471→
   472→   472→      // Check target platform based on active mode
   473→   473→      const rpcConfig = await db.rpcConfig.findFirst({ where: { userId } })
   474→   474→      const isRpcActive = !!(session.rpcEnabled && rpcConfig?.enabled)
   475→   475→      const isStatusActive = !!session.statusEnabled
   476→   476→
   477→   477→      const activePlatform = isStatusActive
   478→   478→        ? (session.statusPlatform || 'mobile')
   479→   479→        : (isRpcActive ? (rpcConfig?.platform || 'desktop') : (session.statusPlatform || 'mobile'))
   480→   480→
   481→   481→      const isQuest = activePlatform === 'meta_quest' || (isStatusActive && session.vrStatusActive)
   482→   482→      const targetPlatform = isQuest ? 'meta_quest' : activePlatform
   483→   483→      userSock.platform = targetPlatform
   484→   484→
   485→   485→      const ws = new WebSocket(CONFIG.discord.gatewayUrl)
   486→   486→      userSock.ws = ws
   487→   487→
   488→   488→      ws.on('open', () => {
   489→   489→        // Awaiting HELLO (OP 10)
   490→   490→      })
   491→   491→
   492→   492→      ws.on('message', async (data: Buffer | string) => {
   493→   493→        try {
   494→   494→          const raw = typeof data === 'string' ? data : data.toString()
   495→   495→          const payload = JSON.parse(raw)
   496→   496→          const op = payload.op
   497→   497→          const t = payload.t
   498→   498→
   499→   499→          if (op === 10) {
   500→   500→            // HELLO: Start heartbeats and send IDENTIFY
   501→   501→            const heartbeatInterval = payload.d?.heartbeat_interval || 41250
   502→   502→            userSock.heartbeatAck = true
   503→   503→
   504→   504→            userSock.heartbeatTimer = setInterval(() => {
   505→   505→              if (ws.readyState === WebSocket.OPEN) {
   506→   506→                if (!userSock.heartbeatAck) {
   507→   507→                  console.warn(`[10X RPC Daemon] Zombie socket detected for user ${userId} (missing ACK). Reconnecting...`)
   508→   508→                  this.cleanupSocket(userSock)
   509→   509→                  this.scheduleReconnect(userId)
   510→   510→                  return
   511→   511→                }
   512→   512→                userSock.heartbeatAck = false
   513→   513→                ws.send(JSON.stringify({ op: 1, d: null }))
   514→   514→              }
   515→   515→            }, heartbeatInterval)
   516→   516→
   517→   517→            const isMobile = targetPlatform === 'android' || targetPlatform === 'ios' || targetPlatform === 'samsung' || targetPlatform === 'mobile'
   518→   518→            const isConsole = targetPlatform === 'console' || targetPlatform === 'xbox' || targetPlatform === 'ps4' || targetPlatform === 'ps5'
   519→   519→            const isWeb = targetPlatform === 'web'
   520→   520→
   521→   521→            const properties = isQuest
   522→   522→              ? { os: 'Android', browser: 'Discord VR', device: 'Meta Quest' }
   523→   523→              : isMobile
   524→   524→              ? {
   525→   525→                  os: targetPlatform === 'ios' ? 'iOS' : 'Android',
   526→   526→                  browser: targetPlatform === 'ios' ? 'Discord iOS' : 'Discord Android',
   527→   527→                  device: targetPlatform === 'ios' ? 'iPhone' : (targetPlatform === 'samsung' ? 'Samsung Galaxy' : 'Android Device'),
   528→   528→                }
   529→   529→              : isConsole
   530→   530→              ? {
   531→   531→                  os: targetPlatform === 'xbox' ? 'Xbox' : 'PlayStation',
   532→   532→                  browser: targetPlatform === 'xbox' ? 'Discord Xbox' : 'Discord PlayStation',
   533→   533→                  device: targetPlatform === 'xbox' ? 'Xbox Series X' : 'PlayStation 5',
   534→   534→                }
   535→   535→              : isWeb
   536→   536→              ? { os: 'Windows', browser: 'Discord Web', device: 'Chrome' }
   537→   537→              : { os: 'Windows', browser: 'Discord Client', device: 'Desktop' }
   538→   538→
   539→   539→            const bearerToken = accessToken.startsWith('Bearer ') ? accessToken : `Bearer ${accessToken}`
   540→   540→            const identify = {
   541→   541→              op: 2,
   542→   542→              d: {
   543→   543→                token: bearerToken,
   544→   544→                properties,
   545→   545→              },
   546→   546→            }
   547→   547→            ws.send(JSON.stringify(identify))
   548→   548→          } else if (op === 11) {
   549→   549→            // Heartbeat ACK
   550→   550→            userSock.heartbeatAck = true
   551→   551→          } else if (op === 1) {
   552→   552→            // Server requested heartbeat
   553→   553→            ws.send(JSON.stringify({ op: 1, d: null }))
   554→   554→          } else if (op === 0 && t === 'READY') {
   555→   555→            // Authenticated and ready! Only trigger initial presence push on READY (NOT on SESSIONS_REPLACE to avoid infinite loop)
   556→   556→            userSock.connected = true
   557→   557→            userSock.isConnecting = false
   558→   558→            userSock.retryCount = 0
   559→   559→            userSock.lastConnectedAt = new Date()
   560→   560→
   561→   561→            await db.session.update({
   562→   562→              where: { id: session.id },
   563→   563→              data: {
   564→   564→                gatewayReady: true,
   565→   565→                lastPresenceUpdate: new Date(),
   566→   566→              },
   567→   567→            }).catch(() => {})
   568→   568→
   569→   569→            // Push current presence immediately
   570→   570→            await this.pushPresenceForUser(userId, session, true)
   571→   571→          } else if (op === 7) {
   572→   572→            // Discord requested reconnect
   573→   573→            console.log(`[10X RPC Daemon] Discord Gateway sent OP 7 RECONNECT for user ${userId}. Reconnecting...`)
   574→   574→            this.cleanupSocket(userSock)
   575→   575→            this.scheduleReconnect(userId, 1000)
   576→   576→          } else if (op === 9) {
   577→   577→            // Invalid session
   578→   578→            console.warn(`[10X RPC Daemon] Discord Gateway sent OP 9 INVALID_SESSION for user ${userId}`)
   579→   579→            this.cleanupSocket(userSock)
   580→   580→            this.scheduleReconnect(userId, 3000)
   581→   581→          }
   582→   582→        } catch (err) {
   583→   583→          console.error(`[10X RPC Daemon] Error handling WS message for user ${userId}:`, err)
   584→   584→        }
   585→   585→      })
   586→   586→
   587→   587→      ws.on('error', (err) => {
   588→   588→        console.error(`[10X RPC Daemon] Gateway WS error for user ${userId}:`, err.message)
   589→   589→        userSock.connected = false
   590→   590→        userSock.isConnecting = false
   591→   591→        this.cleanupSocket(userSock)
   592→   592→        this.scheduleReconnect(userId)
   593→   593→      })
   594→   594→
   595→   595→      ws.on('close', async (code, reason) => {
   596→   596→        console.log(`[10X RPC Daemon] Gateway WS closed for user ${userId} (code: ${code}, reason: ${reason.toString() || 'none'})`)
   597→   597→        userSock.connected = false
   598→   598→        userSock.isConnecting = false
   599→   599→        this.cleanupSocket(userSock)
   600→   600→
   601→   601→        if (code === 4004) {
   602→   602→          // Auth failed — try refreshing token
   603→   603→          console.log(`[10X RPC Daemon] Auth failed (4004) for user ${userId}. Refreshing token...`)
   604→   604→          if (session.discordRefreshToken) {
   605→   605→            const refreshed = await refreshDiscordToken(session.discordRefreshToken)
   606→   606→            if (refreshed) {
   607→   607→              await db.session.update({
   608→   608→                where: { id: session.id },
   609→   609→                data: {
   610→   610→                  discordAccessToken: refreshed.access_token,
   611→   611→                  discordRefreshToken: refreshed.refresh_token,
   612→   612→                  discordTokenExpiresAt: new Date(Date.now() + (refreshed.expires_in || 604800) * 1000),
   613→   613→                },
   614→   614→              })
   615→   615→              this.scheduleReconnect(userId, 2000)
   616→   616→              return
   617→   617→            }
   618→   618→          }
   619→   619→        } else if (code === 4008) {
   620→   620→          // Rate limited — back off for 60 seconds to allow rate limit window to clear
   621→   621→          console.warn(`[10X RPC Daemon] Rate limited by Discord Gateway for user ${userId}. Backing off for 60s...`)
   622→   622→          this.scheduleReconnect(userId, 60000)
   623→   623→          return
   624→   624→        }
   625→   625→
   626→   626→        this.scheduleReconnect(userId)
   627→   627→      })
   628→   628→    } catch (err) {
   629→   629→      console.error(`[10X RPC Daemon] Failed to initialize connection for user ${userId}:`, err)
   630→   630→      userSock.connected = false
   631→   631→      userSock.isConnecting = false
   632→   632→      this.scheduleReconnect(userId)
   633→   633→    }
   634→   634→  }

   636→   636→  /**
   637→   637→   * Schedule reconnection with exponential backoff.
   638→   638→   */
   639→   639→  private scheduleReconnect(userId: string, customDelayMs?: number): void {
   640→   640→    const userSock = this.sockets.get(userId)
   641→   641→    if (!userSock || userSock.retryTimer) return
   642→   642→
   643→   643→    const delay = customDelayMs ?? Math.min(30000, 1000 * Math.pow(1.5, Math.min(userSock.retryCount, 8)))
   644→   644→    userSock.retryCount++
   645→   645→
   646→   646→    userSock.retryTimer = setTimeout(() => {
   647→   647→      userSock.retryTimer = undefined
   648→   648→      this.connectUserSocket(userId)
   649→   649→    }, delay)
   650→   650→  }

   652→   652→  /**
   653→   653→   * Build activities and send OP 3 for the user if anything changed or on reconnect.
   654→   654→   */
   655→   655→  private async pushPresenceForUser(userId: string, session: any, force: boolean = false): Promise<void> {
   656→   656→    const userSock = this.sockets.get(userId)
   657→   657→    if (!userSock || !userSock.ws || userSock.ws.readyState !== WebSocket.OPEN) return
   658→   658→
   659→   659→    let rpcConfig = null
   660→   660→    let globalConfig = null
   661→   661→    try {
   662→   662→      [rpcConfig, globalConfig] = await Promise.all([
   663→   663→        db.rpcConfig.findFirst({ where: { userId } }),
   664→   664→        db.globalConfig.findUnique({ where: { userId } }),
   665→   665→      ])
   666→   666→    } catch (dbErr) {
   667→   667→      console.warn(`[10X RPC Daemon] Transient DB error in pushPresenceForUser for user ${userId}:`, dbErr)
   668→   668→      return
   669→   669→    }
   670→   670→
   671→   671→    const placeholderCtx: PlaceholderContext = {
   672→   672→      timezone: globalConfig?.timezone || 'UTC',
   673→   673→      city: globalConfig?.city || undefined,
   674→   674→      rpcStartedAt: rpcConfig?.updatedAt
   675→   675→        ? new Date(rpcConfig.updatedAt).getTime()
   676→   676→        : (session.lastPresenceUpdate ? new Date(session.lastPresenceUpdate).getTime() : Date.now()),
   677→   677→    }
   678→   678→
   679→   679→    // DATABASE IS SINGLE SOURCE OF TRUTH:
   680→   680→    // Only send RPC if rpcConfig exists AND rpcConfig.enabled === true AND session.rpcEnabled === true
   681→   681→    const isRpcActive = !!(session.rpcEnabled && rpcConfig?.enabled)
   682→   682→    const isStatusActive = !!session.statusEnabled
   683→   683→
   684→   684→    const activePlatform = isStatusActive
   685→   685→      ? (session.statusPlatform || 'mobile')
   686→   686→      : (isRpcActive ? (rpcConfig?.platform || 'desktop') : (session.statusPlatform || 'mobile'))
   687→   687→    userSock.platform = activePlatform === 'meta_quest' ? 'meta_quest' : activePlatform
   688→   688→
   689→   689→    const activities = await buildPresenceActivities({
   690→   690→      rpcConfig: isRpcActive ? rpcConfig : null,
   691→   691→      customStatus: isStatusActive ? session.customStatus : null,
   692→   692→      customStatusEmoji: isStatusActive ? session.customStatusEmoji : null,
   693→   693→      placeholderCtx,
   694→   694→      vrStatusActive: (isStatusActive && session.statusPlatform === 'meta_quest') || (isRpcActive && rpcConfig?.platform === 'meta_quest'),
   695→   695→      platform: isRpcActive ? (rpcConfig?.platform || 'desktop') : (session.statusPlatform || 'mobile'),
   696→   696→    })
   697→   697→
   698→   698→    const status = isStatusActive ? (session.userStatus || 'online') : (isRpcActive ? 'online' : 'invisible')
   699→   699→    const activitiesHash = JSON.stringify({ status, activities })
   700→   700→
   701→   701→    // Avoid spamming identical OP 3 payloads unless forced (prevents rate limits)
   702→   702→    if (!force && userSock.lastActivitiesHash === activitiesHash && userSock.lastStatus === status) {
   703→   703→      return
   704→   704→    }
   705→   705→
   706→   706→    // If socket is open and payload is ready, send OP 3
   707→   707→    if (!userSock.ws || userSock.ws.readyState !== WebSocket.OPEN) return
   708→   708→
   709→   709→    try {
   710→   710→      console.log(`[10X RPC Daemon] Sending OP 3 for user ${userId}: status=${status}, activities=${JSON.stringify(activities)}`)
   711→   711→      userSock.ws.send(JSON.stringify({
   712→   712→        op: 3,
   713→   713→        d: {
   714→   714→          status,
   715→   715→          activities,
   716→   716→          afk: false,
   717→   717→          since: null,
   718→   718→        },
   719→   719→      }))
   720→   720→      userSock.lastStatus = status
   721→   721→      userSock.lastActivitiesHash = activitiesHash
   722→   722→    } catch (err) {
   723→   723→      console.error(`[10X RPC Daemon] Failed to send OP 3 for user ${userId}:`, err)
   724→   724→    }
   725→   725→  }

   734→   734→    this.cleanupSocket(userSock)

   739→   739→  /**
   740→   740→   * Clean up a socket and its timers.
   741→   741→   */
   742→   742→  private cleanupSocket(userSock: ActiveUserSocket): void {
   743→   743→    if (userSock.heartbeatTimer) {
   744→   744→      clearInterval(userSock.heartbeatTimer)
   745→   745→      userSock.heartbeatTimer = undefined
   746→   746→    }
   747→   747→    if (userSock.retryTimer) {
   748→   748→      clearTimeout(userSock.retryTimer)
   749→   749→      userSock.retryTimer = undefined
   750→   750→    }
   751→   751→    if (userSock.ws) {
   752→   752→      try {
   753→   753→        userSock.ws.removeAllListeners()
   754→   754→        userSock.ws.close()
   755→   755→      } catch {}
   756→   756→      userSock.ws = null
   757→   757→    }
   758→   758→    userSock.connected = false
   759→   759→    userSock.isConnecting = false
   760→   760→  }
