    46→    46→export function RichPresenceForm({

    53→    53→}) {
    54→    54→  const [cfg, setCfg] = useState<RpcConfig>(initial || DEFAULT_CONFIG)
    55→    55→  const [saving, setSaving] = useState(false)
    56→    56→  const [enabled, setEnabled] = useState(initial?.enabled ?? false)
    57→    57→  const [diagnosing, setDiagnosing] = useState(false)
    58→    58→  const [diagnoseResult, setDiagnoseResult] = useState<DiagnoseResult | null>(null)
    59→    59→
    60→    60→  const handleDiagnose = async () => {
    61→    61→    setDiagnosing(true)
    62→    62→    setDiagnoseResult(null)
    63→    63→    try {
    64→    64→      const res = await fetch('/api/rpc/diagnose', { cache: 'no-store' })
    65→    65→      const data: DiagnoseResult = await res.json()
    66→    66→      setDiagnoseResult(data)
    67→    67→      const failing = data.checks.filter(c => !c.ok)
    68→    68→      if (failing.length === 0) {
    69→    69→        toast.success('All checks passed — RPC should be working', { duration: 4000 })
    70→    70→      } else {
    71→    71→        toast.error(`${failing.length} issue(s) found — tap the report to view`, { duration: 6000 })
    72→    72→      }
    73→    73→    } catch (e) {
    74→    74→      toast.error('Diagnose failed: ' + (e instanceof Error ? e.message : 'unknown'))
    75→    75→    } finally {
    76→    76→      setDiagnosing(false)
    77→    77→    }
    78→    78→  }
    79→    79→
    80→    80→  const [platformOpen, setPlatformOpen] = useState(false)
    81→    81→  const [typeOpen, setTypeOpen] = useState(false)
    82→    82→  const platformRef = useRef<HTMLDivElement>(null)

    89→    89→        setPlatformOpen(false)
    90→    90→      }
    91→    91→      if (typeRef.current && !typeRef.current.contains(target)) {
    92→    92→        setTypeOpen(false)
    93→    93→      }
    94→    94→    }
    95→    95→    document.addEventListener('mousedown', handleClickOutside)
    96→    96→    return () => document.removeEventListener('mousedown', handleClickOutside)
    97→    97→  }, [])
    98→    98→
    99→    99→  useEffect(() => {

   101→   101→      setCfg({

   103→   103→        type: initial.type ?? 'PLAYING',
   104→   104→        platform: initial.platform ?? 'desktop',
   105→   105→        state: initial.state ?? '',
   106→   106→        details: initial.details ?? '',
   107→   107→        largeImage: initial.largeImage ?? '',
   108→   108→        largeText: initial.largeText ?? '',
   109→   109→        smallImage: initial.smallImage ?? '',
   110→   110→        smallText: initial.smallText ?? '',
   111→   111→        button1Label: initial.button1Label ?? '',
   112→   112→        button1Url: initial.button1Url ?? '',
   113→   113→        button2Label: initial.button2Label ?? '',

   117→   117→        partyId: initial.partyId ?? '',
   118→   118→        partySecret: initial.partySecret ?? '',

   147→   147→      errors.push('Party size cannot exceed party max')
   148→   148→    }

   150→   150→    if (cfg.endTotalMins != null && cfg.endTotalMins < 1) errors.push('End time must be at least 1 minute')
   151→   151→    if (errors.length > 0) {
   152→   152→      errors.forEach(e => toast.error(e))
   153→   153→      return
   154→   154→    }
   155→   155→
   156→   156→    setSaving(true)
   157→   157→    try {
   158→   158→      const res = await api.rpcSave({ ...cfg, enabled })
   159→   159→      if (res.ok) {
   160→   160→        toast.success(enabled ? 'Rich Presence updated & live on Discord' : 'Configuration saved (RPC is OFF)', { duration: 2500 })
   161→   161→        onSaved?.()

   163→   163→        toast.error('Failed to save configuration')
   164→   164→      }
   165→   165→    } catch (e) {
   166→   166→      console.error(e)
   167→   167→      toast.error('Failed to save Rich Presence')
   168→   168→    } finally {
   169→   169→      setSaving(false)
   170→   170→    }
   171→   171→  }

   181→   181→      } else {
   182→   182→        setEnabled(!v)
   183→   183→        toast.error('Failed to toggle RPC')
   184→   184→      }
   185→   185→    } catch (e) {
   186→   186→      console.error(e)
   187→   187→      setEnabled(!v)
   188→   188→      toast.error('Failed to toggle RPC')
   189→   189→    }

   228→   228→                        onClick={() => { set('type', t.label); setTypeOpen(false) }}
   229→   229→                        className={`w-full flex items-center gap-2 px-3 py-2.5 rounded-xl text-xs font-medium uppercase transition-colors text-left ${
   230→   230→                          isSelected
   231→   231→                            ? 'bg-[#6b21a8] text-white font-semibold'
   232→   232→                            : 'text-white/80 hover:text-white hover:bg-white/5'
   233→   233→                        }`}
   234→   234→                      >

   236→   236→                      </button>
